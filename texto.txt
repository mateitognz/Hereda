function Badge({ color, children }) {
  return (
    <span style={{
      background: color + "20",
      color: color,
      border: `1px solid ${color}40`,
      padding: "2px 10px",
      borderRadius: 20,
      fontSize: 11,
      fontWeight: 700,
      letterSpacing: 0.5,
      textTransform: "uppercase",
    }}>
      {children}
    </span>
  );
}

function ProgressBar({ value, max, color = "#B8960C" }) {
  const pct = Math.round((value / max) * 100);
  return (
    <div style={{ background: "#E5E0D5", borderRadius: 4, height: 6, overflow: "hidden" }}>
      <div style={{ width: `${pct}%`, background: color, height: "100%", borderRadius: 4, transition: "width 0.6s ease" }} />
    </div>
  );
}

function StatCard({ label, value, sub, color = "#B8960C", icon }) {
  return (
    <div style={{
      background: "#fff",
      border: "1px solid #E5E0D5",
      borderRadius: 12,
      padding: "20px 24px",
      display: "flex",
      flexDirection: "column",
      gap: 4,
      position: "relative",
      overflow: "hidden",
    }}>
      <div style={{ position: "absolute", right: 16, top: 16, fontSize: 28, opacity: 0.15 }}>{icon}</div>
      <span style={{ fontSize: 11, color: "#888", fontWeight: 600, textTransform: "uppercase", letterSpacing: 1 }}>{label}</span>

      {sub && <span style={{ fontSize: 12, color: "#999" }}>{sub}</span>}

  );
}

// ─── ONBOARDING WIZARD ────────────────────────────────────────────────────────
function OnboardingWizard({ onComplete }) {
  const [step, setStep] = useState(0);
  const [answers, setAnswers] = useState({});
  const [resultado, setResultado] = useState(null);
  const [typing, setTyping] = useState(false);

  const preguntas = [
    {
      id: "tipo",
      pregunta: "¿Qué tipo de trámite necesita realizar?",
      opciones: ["Sucesión intestada (sin testamento)", "Sucesión testada (con testamento)", "No sé / Necesito orientación"],
    },
    {
      id: "herederos",
      pregunta: "¿Cuántos herederos hay?",
      opciones: ["1 heredero", "2 herederos", "3-5 herederos", "Más de 5 herederos"],
    },
    {
      id: "testamento",
      pregunta: "¿Existe testamento notariado?",
      opciones: ["Sí, existe testamento", "No existe testamento", "No estoy seguro/a"],
    },
    {
      id: "bienes",
      pregunta: "¿Qué bienes están involucrados?",
      tipo: "multi",
      opciones: ["Inmueble (casa/apartamento/lote)", "Vehículo", "Cuentas bancarias", "Acciones o inversiones", "Solo bienes muebles", "No hay bienes conocidos"],
    },
    {
      id: "ciudad",
      pregunta: "¿En qué ciudad se encuentran los bienes principales?",
      opciones: ["Bogotá", "Medellín", "Cali", "Barranquilla", "Otra ciudad"],
    },
  ];

  const handleAnswer = (val) => {
    const q = preguntas[step];
    if (q.tipo === "multi") {
      const prev = answers[q.id] || [];
      const next = prev.includes(val) ? prev.filter((x) => x !== val) : [...prev, val];
      setAnswers({ ...answers, [q.id]: next });
    } else {
      const newAnswers = { ...answers, [q.id]: val };
      setAnswers(newAnswers);
      setTyping(true);
      setTimeout(() => {
        setTyping(false);
        if (step < preguntas.length - 1) {
          setStep(step + 1);
        } else {
          generarResultado(newAnswers);
        }
      }, 600);
    }
  };

  const handleMultiNext = () => {
    setTyping(true);
    setTimeout(() => {
      setTyping(false);
      if (step < preguntas.length - 1) setStep(step + 1);
      else generarResultado(answers);
    }, 600);
  };

  const generarResultado = (ans) => {
    const herederosCnt = ans.herederos?.includes("1") ? 1 : ans.herederos?.includes("2") ? 2 : ans.herederos?.includes("3-5") ? 4 : 6;
    const tieneTestamento = ans.testamento?.includes("Sí");
    const bienes = Array.isArray(ans.bienes) ? ans.bienes : [ans.bienes];
    const tieneInmueble = bienes?.some((b) => b?.includes("Inmueble"));

    const docs = [
      "Registro civil de defunción del causante",
      "Cédulas de ciudadanía de todos los herederos",
      "Registros civiles de nacimiento de herederos",
      "Certificado de tradición y libertad (si hay inmueble)",
    ];
    if (tieneTestamento) docs.push("Copia autenticada del testamento");
    if (!tieneTestamento) docs.push("Declaración de ausencia de testamento ante notario");
    if (tieneInmueble) docs.push("Paz y salvo predial del inmueble", "Avalúo catastral vigente");
    if (herederosCnt > 2) docs.push("Poder especial (si algún heredero no puede asistir)");

    const tiempoBase = tieneTestamento ? 45 : 60;
    const tiempoExtra = tieneInmueble ? 15 : 0;
    const tiempoTotal = tiempoBase + tiempoExtra + (herederosCnt > 3 ? 15 : 0);

    const costoBase = tieneInmueble ? 800000 : 400000;
    const costoHerederos = herederosCnt * 50000;

    setResultado({
      docs,
      tiempo: `${tiempoTotal} días hábiles aproximadamente`,
      costo: `$${(costoBase + costoHerederos).toLocaleString("es-CO")} - $${(costoBase + costoHerederos + 500000).toLocaleString("es-CO")} COP`,
      pasos: etapas.map((e) => e.label),
      tipo: tieneTestamento ? "Sucesión Testada" : "Sucesión Intestada",
    });
  };

  const q = preguntas[step];

  return (
    <div style={{ minHeight: "100%", display: "flex", flexDirection: "column", gap: 0 }}>
      {/* Header */}
      <div style={{ padding: "32px 32px 0", borderBottom: "1px solid #E5E0D5", paddingBottom: 24 }}>
        <div style={{ display: "flex", alignItems: "center", gap: 12, marginBottom: 16 }}>
          <div style={{ width: 40, height: 40, background: "#B8960C", borderRadius: 10, display: "flex", alignItems: "center", justifyContent: "center", fontSize: 18 }}>🏛️</div>
          <div>
            <div style={{ fontWeight: 800, fontSize: 18, color: "#0D0D0D" }}>Asistente Inteligente</div>
            <div style={{ fontSize: 12, color: "#888" }}>Notaría 27 · Le ayudamos paso a paso</div>
          </div>
        </div>
        {!resultado && (
          <div style={{ display: "flex", gap: 4 }}>
            {preguntas.map((_, i) => (
              <div key={i} style={{
                flex: 1, height: 3, borderRadius: 2,
                background: i <= step ? "#B8960C" : "#E5E0D5",
                transition: "background 0.3s",
              }} />
            ))}
          </div>
        )}
      </div>

      <div style={{ flex: 1, overflowY: "auto", padding: "24px 32px", display: "flex", flexDirection: "column", gap: 16 }}>
        {resultado ? (
          // RESULTADO
          <div style={{ display: "flex", flexDirection: "column", gap: 20 }}>
            <div style={{ background: "linear-gradient(135deg, #B8960C10, #B8960C05)", border: "1px solid #B8960C30", borderRadius: 16, padding: 24 }}>
              <div style={{ fontSize: 13, color: "#B8960C", fontWeight: 700, marginBottom: 4 }}>ANÁLISIS COMPLETADO</div>
              <div style={{ fontSize: 22, fontWeight: 800, color: "#0D0D0D" }}>{resultado.tipo}</div>
            </div>

            <div style={{ display: "grid", gridTemplateColumns: "1fr 1fr", gap: 12 }}>
              <div style={{ background: "#fff", border: "1px solid #E5E0D5", borderRadius: 12, padding: 16 }}>
                <div style={{ fontSize: 11, color: "#888", fontWeight: 600, marginBottom: 4 }}>⏱ TIEMPO ESTIMADO</div>
                <div style={{ fontWeight: 700, color: "#0D0D0D", fontSize: 15 }}>{resultado.tiempo}</div>
              </div>
              <div style={{ background: "#fff", border: "1px solid #E5E0D5", borderRadius: 12, padding: 16 }}>
                <div style={{ fontSize: 11, color: "#888", fontWeight: 600, marginBottom: 4 }}>💰 COSTO ESTIMADO</div>
                <div style={{ fontWeight: 700, color: "#27AE60", fontSize: 14 }}>{resultado.costo}</div>
              </div>
            </div>

            <div style={{ background: "#fff", border: "1px solid #E5E0D5", borderRadius: 12, padding: 20 }}>
              <div style={{ fontSize: 12, color: "#888", fontWeight: 700, marginBottom: 12, textTransform: "uppercase" }}>📋 Documentos Requeridos</div>
              {resultado.docs.map((d, i) => (
                <div key={i} style={{ display: "flex", alignItems: "flex-start", gap: 8, padding: "8px 0", borderBottom: i < resultado.docs.length - 1 ? "1px solid #F0EDE5" : "none" }}>
                  <span style={{ color: "#B8960C", marginTop: 1 }}>•</span>
                  <span style={{ fontSize: 14, color: "#333" }}>{d}</span>
                </div>
              ))}
            </div>

            <div style={{ background: "#fff", border: "1px solid #E5E0D5", borderRadius: 12, padding: 20 }}>
              <div style={{ fontSize: 12, color: "#888", fontWeight: 700, marginBottom: 12, textTransform: "uppercase" }}>🗺 Proceso Paso a Paso</div>
              {resultado.pasos.map((p, i) => (
                <div key={i} style={{ display: "flex", alignItems: "center", gap: 12, padding: "8px 0" }}>
                  <div style={{ width: 24, height: 24, borderRadius: "50%", background: "#B8960C", color: "#fff", fontSize: 11, fontWeight: 800, display: "flex", alignItems: "center", justifyContent: "center", flexShrink: 0 }}>{i + 1}</div>
                  <span style={{ fontSize: 14, color: "#333" }}>{p}</span>
                </div>
              ))}
            </div>

            <button
              onClick={onComplete}
              style={{ width: "100%", padding: "14px 0", background: "#B8960C", color: "#fff", border: "none", borderRadius: 10, fontWeight: 800, fontSize: 15, cursor: "pointer", letterSpacing: 0.5 }}
            >
              Iniciar Mi Trámite →
            </button>
          </div>
        ) : (
          // PREGUNTA
          <>
            <div style={{ background: "#fff", border: "1px solid #E5E0D5", borderRadius: 16, padding: 20 }}>
              <div style={{ fontSize: 11, color: "#B8960C", fontWeight: 700, marginBottom: 8 }}>Pregunta {step + 1} de {preguntas.length}</div>
              <div style={{ fontSize: 17, fontWeight: 700, color: "#0D0D0D", lineHeight: 1.4 }}>{q.pregunta}</div>
            </div>

            {typing ? (
              <div style={{ display: "flex", gap: 4, padding: "12px 16px", background: "#F5F0E8", borderRadius: 12, width: "fit-content" }}>
                {[0, 1, 2].map((i) => (
                  <div key={i} style={{
                    width: 8, height: 8, borderRadius: "50%", background: "#B8960C",
                    animation: `bounce 0.8s ease-in-out ${i * 0.15}s infinite`,
                  }} />
                ))}
              </div>
            ) : (
              <div style={{ display: "flex", flexDirection: "column", gap: 8 }}>
                {q.opciones.map((op) => {
                  const isSelected = q.tipo === "multi" && (answers[q.id] || []).includes(op);
                  return (
                    <button
                      key={op}
                      onClick={() => handleAnswer(op)}
                      style={{
                        padding: "12px 16px",
                        background: isSelected ? "#B8960C10" : "#fff",
                        border: `1.5px solid ${isSelected ? "#B8960C" : "#E5E0D5"}`,
                        borderRadius: 10,
                        textAlign: "left",
                        cursor: "pointer",
                        fontSize: 14,
                        color: isSelected ? "#B8960C" : "#333",
                        fontWeight: isSelected ? 700 : 400,
                        transition: "all 0.15s",
                      }}
                    >
                      {isSelected ? "✓ " : ""}{op}
                    </button>
                  );
                })}
                {q.tipo === "multi" && (
                  <button
                    onClick={handleMultiNext}
                    disabled={!(answers[q.id]?.length > 0)}
                    style={{ padding: "12px 16px", background: "#B8960C", color: "#fff", border: "none", borderRadius: 10, cursor: "pointer", fontWeight: 700, marginTop: 4 }}
                  >
                    Continuar →
                  </button>
                )}
              </div>
            )}
          </>
        )}
      </div>
    </div>
  );
}

// ─── TRACKER ──────────────────────────────────────────────────────────────────
function TramiteTracker({ tramite }) {
  return (
    <div style={{ display: "flex", flexDirection: "column", gap: 20 }}>
      {/* Header */}
      <div style={{ background: "linear-gradient(135deg, #0D0D0D, #2C3E50)", borderRadius: 16, padding: 24, color: "#fff" }}>
        <div style={{ fontSize: 12, color: "#B8960C", fontWeight: 700, marginBottom: 4 }}>EXPEDIENTE {tramite.id}</div>
        <div style={{ fontSize: 20, fontWeight: 800, marginBottom: 8 }}>{tramite.tipo}</div>
        <div style={{ display: "flex", gap: 16, fontSize: 13, color: "#aaa" }}>
          <span>👤 {tramite.ciudadano}</span>
          <span>📅 Iniciado: {tramite.creado}</span>
          <span>⏱ Est.: {tramite.estimado}</span>
        </div>
      </div>

      {/* Progreso */}
      <div style={{ background: "#fff", border: "1px solid #E5E0D5", borderRadius: 16, padding: 24 }}>
        <div style={{ fontSize: 12, color: "#888", fontWeight: 700, marginBottom: 20, textTransform: "uppercase" }}>Progreso del Trámite</div>
        <div style={{ display: "flex", flexDirection: "column", gap: 0 }}>
          {etapas.map((etapa, idx) => {
            const done = idx < tramite.etapa - 1;
            const current = idx === tramite.etapa - 1;
            const pending = idx > tramite.etapa - 1;
            return (
              <div key={etapa.id} style={{ display: "flex", alignItems: "flex-start", gap: 16, position: "relative" }}>
                <div style={{ display: "flex", flexDirection: "column", alignItems: "center" }}>
                  <div style={{
                    width: 36, height: 36, borderRadius: "50%", flexShrink: 0,
                    background: done ? "#27AE60" : current ? "#B8960C" : "#E5E0D5",
                    border: current ? "3px solid #B8960C40" : "none",
                    display: "flex", alignItems: "center", justifyContent: "center",
                    fontSize: done ? 14 : 12,
                    color: done || current ? "#fff" : "#999",
                    fontWeight: 800,
                    transition: "all 0.3s",
                    boxShadow: current ? "0 0 0 4px #B8960C20" : "none",
                  }}>
                    {done ? "✓" : etapa.icon}
                  </div>
                  {idx < etapas.length - 1 && (
                    <div style={{ width: 2, height: 32, background: done ? "#27AE60" : "#E5E0D5", marginTop: 2 }} />
                  )}
                </div>
                <div style={{ paddingTop: 6, paddingBottom: 16 }}>
                  <div style={{ fontWeight: current ? 800 : 600, color: pending ? "#CCC" : "#0D0D0D", fontSize: 14 }}>
                    {etapa.label}
                  </div>
                  {current && (
                    <div style={{ fontSize: 12, color: "#B8960C", fontWeight: 600, marginTop: 2 }}>
                      ← En proceso · {tramite.responsable}
                    </div>
                  )}
                  {done && (
                    <div style={{ fontSize: 12, color: "#27AE60", fontWeight: 600, marginTop: 2 }}>Completado</div>
                  )}
                </div>
              </div>
            );
          })}
        </div>
      </div>

      {/* Documentos */}
      <div style={{ background: "#fff", border: "1px solid #E5E0D5", borderRadius: 16, padding: 24 }}>
        <div style={{ fontSize: 12, color: "#888", fontWeight: 700, marginBottom: 16, textTransform: "uppercase" }}>Documentos</div>
        <div style={{ display: "grid", gridTemplateColumns: "repeat(4, 1fr)", gap: 12, marginBottom: 16 }}>
          {[
            { label: "Total", val: tramite.documentos.total, color: "#2980B9" },
            { label: "Subidos", val: tramite.documentos.subidos, color: "#8E44AD" },
            { label: "Aprobados", val: tramite.documentos.aprobados, color: "#27AE60" },
            { label: "Rechazados", val: tramite.documentos.rechazados, color: "#C0392B" },
          ].map((d) => (
            <div key={d.label} style={{ textAlign: "center", padding: 12, background: d.color + "10", borderRadius: 10, border: `1px solid ${d.color}20` }}>
              <div style={{ fontSize: 24, fontWeight: 800, color: d.color }}>{d.val}</div>
              <div style={{ fontSize: 11, color: "#888", fontWeight: 600 }}>{d.label}</div>
            </div>
          ))}
        </div>
        <ProgressBar value={tramite.documentos.aprobados} max={tramite.documentos.total} color="#27AE60" />
        <div style={{ fontSize: 12, color: "#888", marginTop: 6 }}>
          {Math.round((tramite.documentos.aprobados / tramite.documentos.total) * 100)}% de documentos aprobados
        </div>
      </div>

      {/* Observaciones */}
      {tramite.observaciones.length > 0 && (
        <div style={{ background: "#FFF8F0", border: "1px solid #E67E2240", borderRadius: 16, padding: 20 }}>
          <div style={{ fontSize: 12, color: "#E67E22", fontWeight: 700, marginBottom: 12, textTransform: "uppercase" }}>⚠ Observaciones Pendientes</div>
          {tramite.observaciones.map((o, i) => (

// ─── CRM KANBAN ───────────────────────────────────────────────────────────────
function CRMKanban({ tramites, onSelect }) {
  const [dragging, setDragging] = useState(null);
  const [columns, setColumns] = useState(() => {
    const cols = {
      nuevo: [],
      documentacion_pendiente: [],
      validacion: [],
      revision_juridica: [],
      elaboracion_escritura: [],
      firma: [],
      entregado: [],
    };
    tramites.forEach((t) => { if (cols[t.estado]) cols[t.estado].push(t); });
    return cols;
  });

  const colDefs = [
    { key: "nuevo", label: "Nuevo", color: "#2980B9" },
    { key: "documentacion_pendiente", label: "Docs. Pendientes", color: "#E67E22" },
    { key: "validacion", label: "En Validación", color: "#8E44AD" },
    { key: "revision_juridica", label: "Revisión Jurídica", color: "#2980B9" },
    { key: "elaboracion_escritura", label: "Escritura", color: "#27AE60" },
    { key: "firma", label: "Firma", color: "#B8960C" },
    { key: "entregado", label: "Entregado ✓", color: "#27AE60" },
  ];

  const handleDrop = (targetKey) => {
    if (!dragging) return;
    const { card, fromKey } = dragging;
    if (fromKey === targetKey) { setDragging(null); return; }
    setColumns((prev) => {
      const next = { ...prev };
      next[fromKey] = prev[fromKey].filter((c) => c.id !== card.id);
      next[targetKey] = [...prev[targetKey], { ...card, estado: targetKey }];
      return next;
    });
    setDragging(null);
  };

  return (
    <div style={{ overflowX: "auto", paddingBottom: 16 }}>
      <div style={{ display: "flex", gap: 12, minWidth: "fit-content" }}>
        {colDefs.map((col) => (
          <div
            key={col.key}
            onDragOver={(e) => e.preventDefault()}
            onDrop={() => handleDrop(col.key)}
            style={{
              width: 220,
              background: "#F8F5F0",
              border: "1px solid #E5E0D5",
              borderRadius: 12,
              overflow: "hidden",
            }}
          >
            <div style={{ padding: "12px 16px", background: col.color + "15", borderBottom: `2px solid ${col.color}` }}>
              <div style={{ fontWeight: 800, fontSize: 12, color: col.color, textTransform: "uppercase" }}>{col.label}</div>
              <div style={{ fontSize: 11, color: "#888", marginTop: 2 }}>{columns[col.key].length} trámites</div>
            </div>
            <div style={{ padding: 8, display: "flex", flexDirection: "column", gap: 8, minHeight: 120 }}>
              {columns[col.key].map((t) => (
                <div
                  key={t.id}
                  draggable
                  onDragStart={() => setDragging({ card: t, fromKey: col.key })}
                  onClick={() => onSelect(t)}
                  style={{
                    background: "#fff",
                    border: "1px solid #E5E0D5",
                    borderRadius: 8,
                    padding: 12,
                    cursor: "grab",
                    transition: "box-shadow 0.15s",
                    boxShadow: "0 1px 3px rgba(0,0,0,0.05)",
                  }}
                >
                  <div style={{ fontSize: 10, color: col.color, fontWeight: 700 }}>{t.id}</div>
                  <div style={{ fontSize: 13, fontWeight: 700, color: "#0D0D0D", marginTop: 4, lineHeight: 1.3 }}>{t.ciudadano}</div>
                  <div style={{ fontSize: 11, color: "#888", marginTop: 4 }}>{t.tipo}</div>
                  <div style={{ marginTop: 8 }}>
                    <ProgressBar value={t.documentos.aprobados} max={t.documentos.total} />
                    <div style={{ fontSize: 10, color: "#888", marginTop: 4 }}>Docs: {t.documentos.aprobados}/{t.documentos.total}</div>
                  </div>
                </div>
              ))}
            </div>
          </div>
        ))}
      </div>
    </div>
  );
}
